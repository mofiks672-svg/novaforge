cmake_minimum_required(VERSION 3.16)
project(NovaForge VERSION 1.5.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# MSVC: static C++ runtime — nova.exe runs without the "VC++ Redistributable"
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(Threads REQUIRED)
add_executable(nova src/main.cpp)
target_link_libraries(nova PRIVATE Threads::Threads)
if(WIN32)
  enable_language(RC)
  target_sources(nova PRIVATE res/nova.rc)                     # icon + version info
  if(MSVC)
    target_sources(nova PRIVATE res/nova.manifest)             # UTF-8 code page, long paths
  else()
    set_source_files_properties(res/nova.rc PROPERTIES COMPILE_DEFINITIONS NOVA_EMBED_MANIFEST)
  endif()
  target_link_libraries(nova PRIVATE ws2_32 shell32 ole32 uuid advapi32)
  target_compile_definitions(nova PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
endif()
if(MSVC)
  target_compile_options(nova PRIVATE /utf-8 /EHsc)
endif()
if(MINGW)
  target_link_options(nova PRIVATE -static)                    # no libstdc++/winpthread DLLs next to nova.exe
endif()
install(TARGETS nova RUNTIME DESTINATION bin)
